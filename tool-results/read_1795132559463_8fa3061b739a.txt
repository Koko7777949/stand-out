    80→function toLocalInput(iso: string): string {
    81→  const d = new Date(iso)
    82→  const pad = (n: number) => String(n).padStart(2, '0')
    83→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
    84→}

   229→function PaymentsTab() {
   230→  const [payments, setPayments] = useState<PaymentRow[]>([])
   231→  const [referrals, setReferrals] = useState<ReviewReferral[]>([])
   232→  const [loading, setLoading] = useState(true)
   233→  const [busyId, setBusyId] = useState<string | null>(null)
   234→
   235→  const load = useCallback(async () => {
   236→    setLoading(true)
   237→    try {
   238→      const r = await api<{ payments: PaymentRow[]; reviewReferrals: ReviewReferral[] }>('/api/admin/payments')
   239→      setPayments(r.payments || [])
   240→      setReferrals(r.reviewReferrals || [])
   241→    } catch (e) {
   242→      toast.error(e instanceof Error ? e.message : 'مشكلة في التحميل')
   243→    } finally {
   244→      setLoading(false)
   245→    }
   246→  }, [])
   247→
   248→  useEffect(() => { load() }, [load])
   249→
   250→  const act = async (id: string, action: 'complete' | 'fail') => {
   251→    setBusyId(id)
   252→    try {
   253→      await api(`/api/admin/payments/${id}`, { body: { action } })
   254→      toast.success(action === 'complete' ? 'اتأكدت الدفعة — التذكرة اتفعّلت للعميل.' : 'الدفعة اترفضت.')
   255→      await load()
   256→    } catch (e) {
   257→      toast.error(e instanceof Error ? e.message : 'مشكلة في التنفيذ')
   258→    } finally {
   259→      setBusyId(null)
   260→    }
   261→  }
   262→
   263→  const reviewReferral = async (id: string, action: 'approve' | 'reject') => {
   264→    setBusyId(id)
   265→    try {
   266→      await api(`/api/admin/referrals/${id}`, { body: { action } })
   267→      toast.success(action === 'approve' ? 'الإحالة اتأكدت والنقاط اتحسبت.' : 'الإحالة اترفضت.')
   268→      await load()
   269→    } catch (e) {
   270→      toast.error(e instanceof Error ? e.message : 'مشكلة في التنفيذ')
   271→    } finally {
   272→      setBusyId(null)
   273→    }
   274→  }
   275→
   276→  if (loading) return <CenterSpinner />
   277→
   278→  return (
   279→    <div className="space-y-6">
   280→      <div className="bg-white nb rounded-2xl p-5">
   281→        <div className="flex items-center justify-between mb-4">
   282→          <p className="font-extrabold text-sm flex items-center gap-1.5">
   283→            <Wallet className="h-4 w-4 text-grape" aria-hidden /> دفعات في انتظار التأكيد ({payments.length})
   284→          </p>
   285→          <Button variant="outline" size="sm" onClick={load} className="gap-1.5 font-bold border-2 border-ink rounded-lg bg-white hover:bg-lilac/50">
   286→            <RefreshCw className="h-3.5 w-3.5" aria-hidden /> تحديث
   287→          </Button>
   288→        </div>
   289→        {payments.length === 0 ? (
   290→          <p className="text-sm font-bold text-muted-foreground py-6 text-center">مفيش دفعات معلّقة — كل حاجة منتظمة.</p>
   291→        ) : (
   292→          <div className="space-y-3 max-h-[480px] overflow-y-auto nice-scroll">
   293→            {payments.map((p) => (
   294→              <div key={p.id} className="rounded-xl border-[2.5px] border-ink bg-lilac/30 p-4 space-y-3">
   295→                <div className="flex flex-wrap items-center justify-between gap-2">
   296→                  <div className="text-sm">
   297→                    <p className="font-extrabold">{p.order.ticket?.holderName || p.order.user.name || 'بدون اسم'}</p>
   298→                    <p className="text-xs font-bold text-muted-foreground" dir="ltr">
   299→                      {p.order.user.phone || p.order.user.email || ''} · {p.order.ticket?.serial || p.order.id.slice(0, 10)}
   300→                    </p>
   301→                  </div>
   302→                  <div className="text-left">
   303→                    <p className="font-extrabold text-grape">{egp(p.amountMinor)}</p>
   304→                    <p className="text-xs font-bold text-muted-foreground">{METHOD_LABELS[p.method] || p.method}</p>
   305→                  </div>
   306→                </div>
